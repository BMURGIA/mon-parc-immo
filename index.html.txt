    // --- BIENS : Charger et Afficher ---
    async function chargerBiens() {
      const { data, error } = await db.from('biens').select('*').order('created_at', { ascending: false });
      if (error) {
        console.error(error);
        return;
      }

      const container = document.getElementById('liste-biens');
      document.getElementById('kpi-biens').innerText = data.length;

      let totalLoyers = 0;
      container.innerHTML = '';

      data.forEach(b => {
        totalLoyers += Number(b.loyer_hc);
        container.innerHTML += `
          <div class="p-4 border rounded-lg bg-slate-50 flex justify-between items-center shadow-sm">
            <div>
              <p class="font-bold text-slate-800">${b.nom}</p>
              <p class="text-xs text-slate-500">${b.adresse} • <span class="bg-slate-200 px-1.5 py-0.5 rounded text-slate-700 font-medium">${b.regime_fiscal}</span></p>

              <p class="text-xs text-slate-400">+${b.charges || 0} € chg</p>
            </div>
          </div>
        `;
      });

      document.getElementById('kpi-loyers').innerText = `${totalLoyers.toFixed(2)} €`;
    }

    // --- BIENS : Ajouter un nouveau bien ---

      e.preventDefault();
      const newBien = {

        adresse: document.getElementById('bien-adresse').value,

      const { error } = await db.from('biens').insert([newBien]);
      if (!error) {
        e.target.reset();

      } else {
        alert('Erreur lors de la sauvegarde : ' + error.message);

    // --- CALCULATEUR IRL ---
    function calculerIndexation() {
      const loyer = parseFloat(document.getElementById('calc-loyer').value);
      const ancien = parseFloat(document.getElementById('calc-ancien-indice').value);
      const nouveau = parseFloat(document.getElementById('calc-nouveau-indice').value);

      if (!loyer || !ancien || !nouveau) {
        alert('Veuillez remplir tous les champs du calculateur.');
        return;
      }

      // Formule Officielle INSEE : Nouveau Loyer = Loyer Actuel x (Nouvel Indice / Ancien Indice)
      const nouveauLoyer = (loyer * (nouveau / ancien)).toFixed(2);
      const augmentation = (nouveauLoyer - loyer).toFixed(2);

      const resDiv = document.getElementById('resultat-indexation');
      resDiv.classList.remove('hidden');
      resDiv.innerHTML = `
        <strong>Nouveau Loyer Révisé : ${nouveauLoyer} € / mois</strong><br>
        <span class="text-xs opacity-80">Augmentation mensuelle de +${augmentation} € (selon la formule légale INSEE)</span>
      `;
    }

    // --- GÉNÉRATEUR QUITTANCE PDF ---
    function genererQuittancePDF() {
      const { jsPDF } = window.jspdf;
      const doc = new jsPDF();

      const bailleur = document.getElementById('pdf-bailleur').value || 'Le Bailleur';
      const locataire = document.getElementById('pdf-locataire').value || 'Le Locataire';
      const bien = document.getElementById('pdf-bien').value || 'Adresse du logement';
      const periode = document.getElementById('pdf-periode').value || 'Mois en cours';
      const montant = document.getElementById('pdf-montant').value || '0';

      // Design du document PDF
      doc.setFont("helvetica", "bold");
      doc.setFontSize(18);
      doc.text("QUITTANCE DE LOYER", 105, 25, { align: "center" });

      doc.setFontSize(10);
      doc.setFont("helvetica", "normal");
      doc.text(`Période : ${periode}`, 105, 33, { align: "center" });

      doc.setLineWidth(0.5);
      doc.line(20, 40, 190, 40);

      doc.text(`Bailleur : ${bailleur}`, 20, 55);
      doc.text(`Locataire : ${locataire}`, 20, 65);
      doc.text(`Adresse du bien : ${bien}`, 20, 75);

      doc.rect(20, 90, 170, 35);
      doc.setFont("helvetica", "bold");
      doc.text(`Montant total perçu : ${montant} €`, 30, 105);
      doc.setFont("helvetica", "normal");
      doc.text(`Je soussigné(e) ${bailleur}, propriétaire du logement désigné ci-dessus,`, 30, 115);
      doc.text(`confirme avoir reçu la somme de ${montant} € au titre du loyer et des charges.`, 30, 120);

      doc.text(`Fait le : ${new Date().toLocaleDateString('fr-FR')}`, 130, 140);
      doc.text("Signature :", 130, 150);

      // Téléchargement automatique
      doc.save(`Quittance_${locataire.replace(/\s+/g, '_')}_${periode}.pdf`);
    }
